cmake_minimum_required(VERSION 3.20)
project(cuda_selector_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Torch REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(selector_parity selector_parity.cpp)
target_include_directories(selector_parity PRIVATE ${REPO})
target_link_libraries(selector_parity PRIVATE ${TORCH_LIBRARIES})
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")

# Phase 1 can be built without CUDA to exercise the unchanged legacy oracle.
option(SELECTOR_CUDA_CANDIDATES "Build experimental CUDA candidates" OFF)
if(SELECTOR_CUDA_CANDIDATES)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_ARCHITECTURES 86)
  find_package(CUDAToolkit 11.8 EXACT REQUIRED)
  add_library(selector_candidates STATIC
    ${REPO}/src/selector_cuda.cpp ${REPO}/src/selector_cuda.cu)
  target_include_directories(selector_candidates PUBLIC ${REPO})
  target_compile_options(selector_candidates PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(selector_candidates PUBLIC ${TORCH_LIBRARIES} CUDA::cudart CUDA::cublas CUDA::cublasLt)
  target_compile_definitions(selector_parity PRIVATE SELECTOR_CUDA_CANDIDATES)
  target_link_libraries(selector_parity PRIVATE selector_candidates)
  add_executable(network_benchmark network_benchmark.cpp)
  target_link_libraries(network_benchmark PRIVATE selector_candidates CUDA::nvToolsExt)
endif()

option(SELECTOR_INTEGRATION_VALIDATION "Link current caller/renderer replay to the isolated mapper library" OFF)
if(SELECTOR_INTEGRATION_VALIDATION)
  find_package(OpenCV REQUIRED)
  find_package(Eigen3 REQUIRED)
  find_package(jsoncpp REQUIRED)
  add_executable(caller_replay caller_replay.cpp)
  target_compile_definitions(caller_replay PRIVATE CARTGS_EXPERIMENTAL_SELECTOR_CUDA)
  target_include_directories(caller_replay PRIVATE ${REPO} ${REPO}/third_party ${OpenCV_INCLUDE_DIRS} /usr/local/cuda-11.8/include)
  target_link_libraries(caller_replay PRIVATE ${REPO}/lib/libgaussian_mapper.so ${REPO}/lib/libcuda_rasterizer.so
    ${TORCH_LIBRARIES} ${OpenCV_LIBRARIES} Eigen3::Eigen jsoncpp_lib /usr/local/cuda-11.8/lib64/libcudart.so
    /usr/local/cuda-11.8/lib64/libnvToolsExt.so)
  add_executable(optimizer_candidate optimizer_candidate.cpp)
  target_include_directories(optimizer_candidate PRIVATE ${REPO} /usr/local/cuda-11.8/include)
  target_link_libraries(optimizer_candidate PRIVATE ${REPO}/lib/libgaussian_mapper.so ${TORCH_LIBRARIES}
    /usr/local/cuda-11.8/lib64/libcudart.so)
  add_executable(guard_transaction guard_transaction.cpp)
  target_compile_definitions(guard_transaction PRIVATE CARTGS_EXPERIMENTAL_SELECTOR_CUDA)
  target_include_directories(guard_transaction PRIVATE ${REPO} /usr/local/cuda-11.8/include)
  target_link_libraries(guard_transaction PRIVATE ${REPO}/lib/libgaussian_mapper.so ${TORCH_LIBRARIES}
    /usr/local/cuda-11.8/lib64/libcudart.so)
  add_executable(operator_trace operator_trace.cpp)
  target_compile_definitions(operator_trace PRIVATE CARTGS_EXPERIMENTAL_SELECTOR_CUDA)
  target_include_directories(operator_trace PRIVATE ${REPO} /usr/local/cuda-11.8/include)
  target_link_libraries(operator_trace PRIVATE ${REPO}/lib/libgaussian_mapper.so ${TORCH_LIBRARIES}
    /usr/local/cuda-11.8/lib64/libcudart.so)
endif()
