cmake_minimum_required(VERSION 3.20)
project(selector_backward_diagnostics LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 86)
find_package(Torch REQUIRED)
find_package(CUDAToolkit 11.8 EXACT REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
add_library(profile_selector STATIC frozen/selector_cuda.cpp profile_selector.cu)
target_include_directories(profile_selector PUBLIC ${REPO})
target_compile_options(profile_selector PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
target_link_libraries(profile_selector PUBLIC ${TORCH_LIBRARIES} CUDA::cudart CUDA::cublas CUDA::cublasLt CUDA::nvToolsExt)
add_executable(backward_profile profile.cpp)
target_link_libraries(backward_profile PRIVATE profile_selector)
add_library(candidate_selector STATIC diagnostic_network.cpp ${REPO}/src/selector_cuda.cu)
target_include_directories(candidate_selector PUBLIC ${REPO})
target_compile_options(candidate_selector PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
target_link_libraries(candidate_selector PUBLIC ${TORCH_LIBRARIES} CUDA::cudart CUDA::cublas CUDA::cublasLt CUDA::nvToolsExt)
add_executable(backward_benchmark benchmark.cpp)
target_link_libraries(backward_benchmark PRIVATE candidate_selector)
add_executable(backward_boundary boundary.cpp)
target_link_libraries(backward_boundary PRIVATE candidate_selector)
add_executable(backward_paired paired.cpp)
target_link_libraries(backward_paired PRIVATE candidate_selector)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(jsoncpp REQUIRED)
add_executable(backward_caller ../cuda_selector_implementation_20261015/caller_replay.cpp)
target_compile_definitions(backward_caller PRIVATE CARTGS_EXPERIMENTAL_SELECTOR_CUDA)
target_include_directories(backward_caller PRIVATE ${REPO}/third_party ${OpenCV_INCLUDE_DIRS})
target_link_libraries(backward_caller PRIVATE candidate_selector ${REPO}/lib/libgaussian_mapper.so
 ${REPO}/lib/libcuda_rasterizer.so ${OpenCV_LIBRARIES} Eigen3::Eigen jsoncpp_lib)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/profile_candidate.cu")
 add_library(candidate_profile STATIC diagnostic_network.cpp profile_candidate.cu)
 target_include_directories(candidate_profile PUBLIC ${REPO})
 target_compile_options(candidate_profile PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
 target_link_libraries(candidate_profile PUBLIC ${TORCH_LIBRARIES} CUDA::cudart CUDA::cublas CUDA::cublasLt CUDA::nvToolsExt)
 add_executable(backward_profile_candidate profile.cpp)
 target_link_libraries(backward_profile_candidate PRIVATE candidate_profile)
endif()
