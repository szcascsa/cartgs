cmake_minimum_required(VERSION 3.16)
project(original_selector_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Torch REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(native_benchmark native_benchmark.cpp ${REPO}/src/protection_profiler.cpp ${REPO}/src/gi_teacher.cpp)
target_include_directories(native_benchmark PRIVATE ${REPO} /usr/local/cuda-11.8/include)
target_link_libraries(native_benchmark ${TORCH_LIBRARIES} /usr/local/cuda-11.8/lib64/libcudart.so /usr/local/cuda-11.8/lib64/libnvToolsExt.so)
target_compile_options(native_benchmark PRIVATE -O3)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(gradient_paths gradient_paths.cpp)
target_include_directories(gradient_paths PRIVATE ${REPO} ${REPO}/third_party ${OpenCV_INCLUDE_DIRS} /usr/local/cuda-11.8/include)
target_link_libraries(gradient_paths ${REPO}/lib/libgaussian_mapper.so ${REPO}/lib/libcuda_rasterizer.so ${TORCH_LIBRARIES} ${OpenCV_LIBRARIES} Eigen3::Eigen /usr/local/cuda-11.8/lib64/libcudart.so)
target_compile_options(gradient_paths PRIVATE -O3)
add_executable(optimizer_benchmark optimizer_benchmark.cpp)
target_include_directories(optimizer_benchmark PRIVATE ${REPO} /usr/local/cuda-11.8/include)
target_link_libraries(optimizer_benchmark ${REPO}/lib/libgaussian_mapper.so ${TORCH_LIBRARIES} /usr/local/cuda-11.8/lib64/libcudart.so)
target_compile_options(optimizer_benchmark PRIVATE -O3)
add_executable(isolated_costs isolated_costs.cpp ${REPO}/src/protection_profiler.cpp ${REPO}/src/gi_teacher.cpp)
target_include_directories(isolated_costs PRIVATE ${REPO} /usr/local/cuda-11.8/include)
target_link_libraries(isolated_costs ${TORCH_LIBRARIES} /usr/local/cuda-11.8/lib64/libcudart.so)
target_compile_options(isolated_costs PRIVATE -O3)
